cmake_minimum_required(VERSION 3.24)
project(WarpMoE VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_library(warpmoe_reference STATIC src/reference.cpp)
target_include_directories(warpmoe_reference PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_options(warpmoe_reference PRIVATE -Wall -Wextra -Wpedantic)
add_executable(reference_tests tests/test_reference.cpp)
target_link_libraries(reference_tests PRIVATE warpmoe_reference)
if(BUILD_TESTING)
  add_test(NAME cpu_reference COMMAND reference_tests)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME roofline_calculation COMMAND ${Python3_EXECUTABLE}
             ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_roofline.py)
  endif()
endif()
option(WARPMOE_REQUIRE_CUDA "Fail configuration if NVIDIA CUDA is unavailable" OFF)
include(CheckLanguage)
check_language(CUDA)
if(CMAKE_CUDA_COMPILER)
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 17)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES 86)
  endif()
  find_package(CUDAToolkit REQUIRED)
  add_library(warpmoe_cuda STATIC cuda/kernels.cu)
  target_include_directories(warpmoe_cuda PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
  set_target_properties(warpmoe_cuda PROPERTIES CUDA_SEPARABLE_COMPILATION OFF)
  target_compile_options(warpmoe_cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
  target_link_libraries(warpmoe_cuda PUBLIC CUDA::cudart)
  add_executable(warpmoe_bench src/cuda_bench.cu)
  target_link_libraries(warpmoe_bench PRIVATE warpmoe_reference warpmoe_cuda CUDA::cudart)
  target_include_directories(warpmoe_bench PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
  add_executable(warpmoe_bandwidth cuda/bandwidth.cu)
  target_link_libraries(warpmoe_bandwidth PRIVATE CUDA::cudart)
else()
  if(WARPMOE_REQUIRE_CUDA)
    message(FATAL_ERROR "CUDA compiler not found. Install CUDA Toolkit or omit WARPMOE_REQUIRE_CUDA.")
  endif()
  message(STATUS "CUDA compiler not detected: building and testing CPU reference only")
endif()
